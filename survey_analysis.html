In [9]:
results = []
n = len(highlights_dummy)

pre_cols = [c for c in highlights_dummy.columns if c.startswith("pre_")]
categories = [c.replace("pre_", "") for c in pre_cols if f"post_{c.replace('pre_', '')}" in highlights_dummy.columns]

for cat in categories:
    pre = highlights_dummy[f"pre_{cat}"]
    post = highlights_dummy[f"post_{cat}"]

    stayed_yes = ((pre == 1) & (post == 1)).sum()
    lost = ((pre == 1) & (post == 0)).sum()
    gained = ((pre == 0) & (post == 1)).sum()
    stayed_no = ((pre == 0) & (post == 0)).sum()

    table = [[stayed_yes, lost],
             [gained, stayed_no]]

    test = mcnemar(table, exact=True)

    results.append({
        "category": cat,
        "n": n,
        "pre_n": pre.sum(),
        "pre_%": pre.mean() * 100,
        "post_n": post.sum(),
        "post_%": post.mean() * 100,
        "change_pp": (post.mean() - pre.mean()) * 100,
        "stayed_yes": stayed_yes,
        "lost_pre_yes_post_no": lost,
        "gained_pre_no_post_yes": gained,
        "stayed_no": stayed_no,
        "mcnemar_p": test.pvalue
    })

stats_results = pd.DataFrame(results)

stats_results["mcnemar_p_holm"] = multipletests(
    stats_results["mcnemar_p"],
    method="holm"
)[1]

stats_results = stats_results.sort_values("change_pp")

    ["category", "mcnemar_p"]
].copy()

    "category": ["error_or_unsure"],

    [holm_results, combined_result],
    ignore_index=True

# Visualisierung mit Übergangsmatrix

# Hilfsfunktion: Mehrfachlabels sauber in Listen umwandeln
def split_labels(x):
    if pd.isna(x) or str(x).strip() == "":
        return ["No highlight"]
    return [label.strip() for label in str(x).split(",")]

# Listen-Spalten erzeugen
highlights["pre_list"] = highlights["highlight_pre"].apply(split_labels)
highlights["post_list"] = highlights["highlight_post"].apply(split_labels)

# Alle Kategorien sammeln
all_categories = sorted(
    set(label for labels in highlights["pre_list"] for label in labels)
    | set(label for labels in highlights["post_list"] for label in labels)
)

# Übergangsmatrix initialisieren
transition_matrix = pd.DataFrame(
    0,
    index=all_categories,
    columns=all_categories
)

# Übergänge zählen
# Bei Mehrfachlabels zählt jede pre-post-Kombination als ein Übergang
for _, row in highlights.iterrows():
    for pre_label in row["pre_list"]:
        for post_label in row["post_list"]:
            transition_matrix.loc[pre_label, post_label] += 1
